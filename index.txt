        function obtenerOperacionesDelMes(anio, mesSel) {
            let ops = [];
            datosGlobales.transacciones.forEach(t => {
                const f = new Date(t.fecha);
                if (f.getFullYear() === anio && (f.getMonth() + 1) === mesSel) {
                    ops.push(t);
                }
            });

            datosGlobales.recurrentes.forEach(r => {
                const fechaFijo = new Date(anio, mesSel - 1, r.dia, 12, 0, 0);
                ops.push({
                    telefono: "Fijo",
                    tipo: "gasto",
                    concepto: r.concepto,
                    categoria: r.categoria || "Vivienda",
                    cantidad: r.cantidad,
                    esFijo: true,
                    fecha: fechaFijo.toISOString()
                });
            });

            return ops;
        }

        function calcularKPIsAnualesMedias() {
            const mesSelStr = document.getElementById('filtroMes').value;
            const [anioSel] = mesSelStr.split('-').map(Number);
            document.getElementById('lblAnioActual').innerText = anioSel;

            let totalIngresosAnio = 0;
            let totalGastosAnio = 0;
            for (let m = 1; m <= 12; m++) {
                let opsM = obtenerOperacionesDelMes(anioSel, m);
                opsM.forEach(t => {
                    if (t.tipo === 'ingreso') totalIngresosAnio += t.cantidad;
                    if (t.tipo === 'gasto') totalGastosAnio += t.cantidad;
                });
            }
            let ahorroAnualTotal = totalIngresosAnio - totalGastosAnio;
            const elAhorroAnual = document.getElementById('kpiAhorroAnual');
            elAhorroAnual.innerText = `${ahorroAnualTotal.toFixed(2)} €`;
            elAhorroAnual.style.color = ahorroAnualTotal >= 0 ? 'var(--success)' : 'var(--danger)';

            let totalIngresos12m = 0;
            let totalGastos12m = 0;
            let acumuladoPorCategoria = {};
            let mesesContados = 0;

            let fechaCursor = new Date(anioSel, parseInt(mesSelStr.split('-')[1]) - 1, 1);
            for (let i = 0; i < 12; i++) {
                let anio = fechaCursor.getFullYear();
                let mes = fechaCursor.getMonth() + 1;
                
                let ops = obtenerOperacionesDelMes(anio, mes);
                if (ops.length > 0) {
                    mesesContados++;
                    ops.forEach(t => {
                        if (t.tipo === 'ingreso') {
                            totalIngresos12m += t.cantidad;
                        } else if (t.tipo === 'gasto') {
                            totalGastos12m += t.cantidad;
                            acumuladoPorCategoria[t.categoria] = (acumuladoPorCategoria[t.categoria] || 0) + t.cantidad;
                        }
                    });
                }
                fechaCursor.setMonth(fechaCursor.getMonth() - 1);
            }

            let divisor = mesesContados > 0 ? mesesContados : 1;
            document.getElementById('kpiMediaIngresos').innerText = `${(totalIngresos12m / divisor).toFixed(2)} €`;
            document.getElementById('kpiMediaGastos').innerText = `${(totalGastos12m / divisor).toFixed(2)} €`;

            const contenedorMedias = document.getElementById('kpiMediaGastoCategorias');
            contenedorMedias.innerHTML = '';
            
            if (Object.keys(acumuladoPorCategoria).length > 0) {
                for (let cat in acumuladoPorCategoria) {
                    let mediaCat = acumuladoPorCategoria[cat] / divisor;
                    contenedorMedias.innerHTML += `
                        <div class="media-cat-item">
                            <span>📂 ${cat}</span>
                            <strong>${mediaCat.toFixed(2)} €/mes</strong>
                        </div>
                    `;
                }
            } else {
                contenedorMedias.innerHTML = '<span style="color: #94a3b8; font-weight: normal;">Sin datos registrados.</span>';
            }
        }

        }

                if(mes > 12) { mes = 1; anio++; }
                let claveMes = `${anio}-${String(mes).padStart(2, '0')}`;
                datosGlobales.metasPorMes[claveMes] = nuevaBaseSugerida;
            }

            alert(`¡Configurado con éxito! Se ha fijado una base de ahorro de ${nuevaBaseSugerida} € durante los próximos ${mesesElegidos} meses.`);
            procesarVistaPrincipal();
        }

        function procesarVistaPrincipal() {
            const mesSeleccionado = document.getElementById('filtroMes').value;
            const [anioSel, mesSel] = mesSeleccionado.split('-').map(Number);

            let totalIngresos = 0;
            let totalGastos = 0;
            let gastosPorCategoriaActual = {};
            
            const lista = document.getElementById('listaMovimientos');
            lista.innerHTML = '';

            const operacionesMes = obtenerOperacionesDelMes(anioSel, mesSel);

            if(operacionesMes.length > 0) {
                operacionesMes.sort((a,b) => new Date(b.fecha) - new Date(a.fecha));

                operacionesMes.forEach(t => {
                    if (t.tipo === 'ingreso') {
                        totalIngresos += t.cantidad;
                    } else {
                        totalGastos += t.cantidad;
                        gastosPorCategoriaActual[t.categoria] = (gastosPorCategoriaActual[t.categoria] || 0) + t.cantidad;
                    }

                    const div = document.createElement('div');
                    div.className = 'movimiento-item';
                    const etiquetaFijo = t.esFijo ? ' 📌 [Gasto Fijo]' : '';
                    div.innerHTML = `
                        <div>
                            <strong>${t.concepto}</strong>${etiquetaFijo}<br>
                            <small style="color:#64748b;">📂 ${t.categoria} · 📅 ${new Date(t.fecha).toLocaleDateString()} · 📱 ${t.telefono}</small>
                        </div>
                        <div style="color: ${t.tipo === 'ingreso' ? 'var(--success)' : 'var(--danger)'}; font-weight: bold;">
                            ${t.tipo === 'ingreso' ? '+' : '-'}${t.cantidad.toFixed(2)} €
                        </div>
                    `;
                    lista.appendChild(div);
                });
            } else {
                lista.innerHTML = '<p style="text-align: center; color: #94a3b8; font-size: 0.9rem;">No hay movimientos en este mes.</p>';
            }

            const balance = totalIngresos - totalGastos;
            document.getElementById('balanceTotal').innerText = `${balance.toFixed(2)} €`;
            document.getElementById('detalleIngresosGastos').innerText = `Ingresos: ${totalIngresos.toFixed(2)}€ | Gastos: ${totalGastos.toFixed(2)}€`;

            const metaInput = document.getElementById('metaAhorroMes').value;
            const meta = metaInput ? parseFloat(metaInput) : 200;
            datosGlobales.metasPorMes[mesSeleccionado] = meta;
            
            let mesAnt = mesSel - 1;
            let anioAnt = anioSel;
            if (mesAnt === 0) { mesAnt = 12; anioAnt--; }

            let gastosPorCategoriaAnterior = {};
            let operacionesMesAnt = obtenerOperacionesDelMes(anioAnt, mesAnt);
            operacionesMesAnt.forEach(t => {
                if (t.tipo === 'gasto') {
                    gastosPorCategoriaAnterior[t.categoria] = (gastosPorCategoriaAnterior[t.categoria] || 0) + t.cantidad;
                }
            });

            let catIncrementoMax = null;
            let mayorDiferencia = 0;

            for (let cat in gastosPorCategoriaActual) {
                let actual = gastosPorCategoriaActual[cat];
                let anterior = gastosPorCategoriaAnterior[cat] || 0;
                let diff = actual - anterior;
                
                if (diff > mayorDiferencia) {
                    mayorDiferencia = diff;
                    catIncrementoMax = cat;
                }
            }

            const bannerCelebracion = document.getElementById('bannerCelebracion');
            const bannerConsejo = document.getElementById('bannerConsejoCategoria');

            let mensajeHTML = "";
            let mostrarBannerConsejo = false;

            if (catIncrementoMax && mayorDiferencia > 0) {
                mostrarBannerConsejo = true;
                let gastoActualCat = gastosPorCategoriaActual[catIncrementoMax];
                mensajeHTML = `📊 Este mes habéis gastado <strong>${gastoActualCat.toFixed(2)} €</strong> en la categoría <strong>"${catIncrementoMax}"</strong> (<strong>${mayorDiferencia.toFixed(2)} € más</strong> que el mes anterior).`;
            }

            if (balance >= meta) {
                bannerCelebracion.style.display = 'block';
            } else {
                bannerCelebracion.style.display = 'none';
                let deficitPendiente = meta - balance;
                
                let avisoDeficit = `⚠️ <strong>Desfase en la meta:</strong> Faltan ${deficitPendiente.toFixed(2)} € para el objetivo.<br><br>
                ¿En cuántos meses queréis resarcir este importe?<br>
                <div style="margin-top: 6px; display: flex; gap: 8px; align-items: center;">
                    <select id="selectMesesRecuperacion" onchange="actualizarProrrateoDinamico(${deficitPendiente})" style="width: 100px; margin:0; padding:4px; background:white;">
                        <option value="2">2 meses</option>
                        <option value="4">4 meses</option>
                        <option value="6" selected>6 meses</option>
                        <option value="8">8 meses</option>
                        <option value="12">12 meses</option>
                    </select>
                    <button onclick="aplicarProrrateoAutomatico(${deficitPendiente})" style="width:auto; padding:6px 12px; font-size:0.8rem;">Aplicar Plan</button>
                </div>
                <div id="textoCuotaCalculada" style="margin-top: 6px; font-size: 0.82rem; font-weight: normal;">
                    Para compensar el desfase en <strong>6 meses</strong>, añadiríais <strong>${(deficitPendiente / 6).toFixed(2)} €/mes</strong> extra a la base actual.
                </div>`;
                
                if (mostrarBannerConsejo) {
                    mensajeHTML += `<br><br>` + avisoDeficit;
                } else {
                    mostrarBannerConsejo = true;
                    mensajeHTML = avisoDeficit;
                }
            }

            if (mostrarBannerConsejo) {
                bannerConsejo.style.display = 'block';
                document.getElementById('textoConsejoCategoria').innerHTML = mensajeHTML;
            } else {
                bannerConsejo.style.display = 'none';
            }
            
            calcularKPIsAnualesMedias();
        }

            const listaRec = document.getElementById('listaRecurrentes');
            listaRec.innerHTML = '';
            if(datosGlobales.recurrentes && datosGlobales.recurrentes.length > 0) {

        function cambiarMesCalendario(dir) {
            mesCalendarioActual.mes += dir;
            if (mesCalendarioActual.mes > 11) { mesCalendarioActual.mes = 0; mesCalendarioActual.anio++; }
            if (mesCalendarioActual.mes < 0) { mesCalendarioActual.mes = 11; mesCalendarioActual.anio--; }
            renderizarCalendario();
        }

        function renderizarCalendario() {
            const anio = mesCalendarioActual.anio;
            const mes = mesCalendarioActual.mes;
            
            const nombresMeses = ["Enero", "Febrero", "Marzo", "Abril", "Mayo", "Junio", "Julio", "Agosto", "Septiembre", "Octubre", "Noviembre", "Diciembre"];
            document.getElementById('tituloMesCalendario').innerText = `${nombresMeses[mes]} ${anio}`;

            const grid = document.getElementById('calendarioGrid');
            grid.innerHTML = '';

            const diasSemana = ['Lu', 'Ma', 'Mi', 'Ju', 'Vi', 'Sa', 'Do'];
            diasSemana.forEach(d => {
                grid.innerHTML += `<div class="calendar-day-name">${d}</div>`;
            });

            let primerDiaIndex = new Date(anio, mes, 1).getDay();
            primerDiaIndex = primerDiaIndex === 0 ? 6 : primerDiaIndex - 1;
            const totalDias = new Date(anio, mes + 1, 0).getDate();

            for (let i = 0; i < primerDiaIndex; i++) {
                grid.innerHTML += `<div class="calendar-day" style="background: transparent; border: none; cursor: default;"></div>`;
            }

            const operacionesMes = obtenerOperacionesDelMes(anio, mes + 1);
            let transaccionesPorDia = {};
            operacionesMes.forEach(t => {
                const f = new Date(t.fecha);
                const diaNum = f.getDate();
                if (!transaccionesPorDia[diaNum]) transaccionesPorDia[diaNum] = [];
                transaccionesPorDia[diaNum].push(t);
            });

            for (let dia = 1; dia <= totalDias; dia++) {
                const transDia = transaccionesPorDia[dia] || [];
                let etiquetaHtml = `<span class="num">${dia}</span>`;
                let claseExtra = transDia.length > 0 ? 'has-events' : '';

                if (transDia.length > 0) {
                    transDia.forEach(t => {
                        let signo = t.tipo === 'ingreso' ? '+' : '-';
                        let claseColor = t.esFijo ? 'event-fijo' : (t.tipo === 'ingreso' ? 'event-ingreso' : 'event-gasto');
                        let prefijoFijo = t.esFijo ? '📌 ' : '';
                        etiquetaHtml += `<div class="calendar-event ${claseColor}">${prefijoFijo}${t.categoria}: ${signo}${t.cantidad}€</div>`;
                    });
                } else {
                    etiquetaHtml += `<span style="opacity:0.3; font-size:0.6rem;">-</span>`;
                }

                const celda = document.createElement('div');
                celda.className = `calendar-day ${claseExtra}`;
                celda.innerHTML = etiquetaHtml;
                celda.onclick = () => abrirModalDia(dia, mes, anio, transDia);
                grid.appendChild(celda);
            }
        }

        function abrirModalDia(dia, mes, anio, transacciones) {
            const nombresMeses = ["Enero", "Febrero", "Marzo", "Abril", "Mayo", "Junio", "Julio", "Agosto", "Septiembre", "Octubre", "Noviembre", "Diciembre"];
            document.getElementById('modalTituloFecha').innerText = `Movimientos del ${dia} de ${nombresMeses[mes]} de ${anio}`;
            
            const contenedorModal = document.getElementById('modalListaMovimientos');
            contenedorModal.innerHTML = '';

            if (transacciones && transacciones.length > 0) {
                transacciones.forEach(t => {
                    const signo = t.tipo === 'ingreso' ? '+' : '-';
                    const colorTexto = t.tipo === 'ingreso' ? 'var(--success)' : 'var(--danger)';
                    const etiquetaFijo = t.esFijo ? ' 📌 [Gasto Fijo]' : '';
                    
                    contenedorModal.innerHTML += `
                        <div class="movimiento-item">
                            <div>
                                <strong>${t.concepto}</strong>${etiquetaFijo}<br>
                                <small style="color:#64748b;">📂 ${t.categoria} · 📱 ${t.telefono}</small>
                            </div>
                            <div style="color: ${colorTexto}; font-weight: bold;">
                                ${signo}${t.cantidad.toFixed(2)} €
                            </div>
                        </div>
                    `;
                });
            } else {
                contenedorModal.innerHTML = `<p style="text-align: center; color: #94a3b8; font-size: 0.9rem; padding: 20px 0;">No hay operaciones registradas en este día.</p>`;
            }

            document.getElementById('modalDia').style.display = 'flex';
        }

        function cerrarModalDia() {
            document.getElementById('modalDia').style.display = 'none';
        }

        function cerrarModalDiaFuera(event) {
            if (event.target.id === 'modalDia') {
                cerrarModalDia();
            }
        }
